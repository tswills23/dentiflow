Diagnosis form — overdue list calculator + qualified Lead event.

WHERE THIS GOES
  GHL → Funnel → Steps → Booking Page → add a Custom HTML/JavaScript block
  directly ABOVE the calendar block.

BEFORE YOU PASTE — set two things in the CONFIG block below:
  WEBHOOK_URL   your GHL Inbound Webhook URL (Workflows → new workflow →
                trigger "Inbound Webhook" → copy the URL it gives you)
  CAL_SELECTOR  the id or class wrapping your calendar. The hero CTA already
                links to #df-cal, so leave it unless the calendar moved.

WHAT IT DOES
  - Collects three numbers that are NOT patient data. No names, no PHI, ever.
  - Calculates a projection from the measured 4.9% booking rate.
  - Shows the result in place, then reveals the calendar underneath it.
  - Fires the standard `Lead` pixel event ONLY when overdue >= 500.
  - POSTs everything to GHL so the contact exists and the SMS sequence can
    merge their own numbers.

FAIL-OPEN BY DESIGN
  The calendar is only hidden after JS confirms it can show it again. If the
  script errors or is blocked, the calendar stays visible and booking still
  works. Never let a form break the only conversion path on the page.

<script>
(function () {
  // ── CONFIG ──────────────────────────────────────────────
  var WEBHOOK_URL  = 'PASTE_YOUR_GHL_INBOUND_WEBHOOK_URL';
  var CAL_SELECTOR = '#df-cal';
  var BOOKING_RATE = 0.049;   // measured: 32 booked / 657 texted
  var MIN_OVERDUE  = 500;     // the gate — below this we capture but do not fire Lead
  // ────────────────────────────────────────────────────────

  var root   = document.getElementById('df-diag');
  if (!root) return;
  var form   = root.querySelector('.df-diag__form');
  var result = root.querySelector('.df-diag__result');
  var errEl  = root.querySelector('.df-diag__err');
  var btn    = root.querySelector('.df-diag__btn');
  var cal    = document.querySelector(CAL_SELECTOR);

  // Fail-open: only hide the calendar once we know we can bring it back.
  // A JS error must never leave the page with no way to book.
  if (cal) { cal.style.display = 'none'; }

  var money = function (n) { return '$' + Math.round(n).toLocaleString('en-US'); };
  var num   = function (n) { return Math.round(n).toLocaleString('en-US'); };

  function fail(msg) {
    errEl.textContent = msg;
    errEl.hidden = false;
    btn.disabled = false;
    btn.querySelector('span').textContent = 'Show me the number';
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    errEl.hidden = true;

    var d = {};
    new FormData(form).forEach(function (v, k) { d[k] = typeof v === 'string' ? v.trim() : v; });

    if (!d.first_name || !d.practice_name || !d.email || !d.phone || !d.pms) {
      return fail('Please fill in every field.');
    }
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(d.email)) return fail('That email looks wrong.');
    if (d.phone.replace(/\D/g, '').length < 10)      return fail('That mobile number looks too short.');

    var overdue = parseInt(d.overdue_count, 10);
    var avg     = parseInt(d.avg_production, 10);
    if (!(overdue >= 0))  return fail('Enter roughly how many patients are overdue.');
    if (!(avg > 0))       return fail('Enter your average production per visit.');

    btn.disabled = true;
    btn.querySelector('span').textContent = 'Working it out…';

    var bookings   = Math.round(overdue * BOOKING_RATE);
    var production = bookings * avg;
    var qualified  = overdue >= MIN_OVERDUE;
    var eventId    = 'lead_' + Date.now() + '_' + Math.floor(Math.random() * 1e6);

    var payload = {
      first_name: d.first_name, practice_name: d.practice_name,
      email: d.email, phone: d.phone, pms: d.pms,
      overdue_count: overdue, avg_production: avg,
      projected_bookings: bookings, projected_production: production,
      qualified: qualified ? 'yes' : 'no',
      event_id: eventId,
      page_url: window.location.href
    };

    // Send to GHL, then render regardless. A webhook hiccup must not cost the
    // visitor their result — we'd rather lose the record than the booking.
    var send = fetch(WEBHOOK_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    }).catch(function () {});

    send.then(function () {
      root.querySelector('.df-o').textContent = num(overdue);
      root.querySelector('.df-b').textContent = num(bookings);
      root.querySelector('.df-p').textContent = money(production);

      root.querySelector('.df-diag__cta-line').textContent = qualified
        ? 'Pick a time below and we’ll walk through where those ' + num(overdue) + ' sit and which segment to work first.'
        : 'The pilot starts at 500 overdue patients, so you’re not there yet — but pick a time if you’d like to talk through how to get the list working.';

      form.hidden   = true;
      result.hidden = false;

      // Standard Lead event, qualified only. Meta should learn one pattern.
      if (qualified && typeof fbq === 'function') {
        fbq('track', 'Lead', { value: production, currency: 'USD' }, { eventID: eventId });
      }

      if (cal) { cal.style.display = ''; }
      result.scrollIntoView({ behavior: 'smooth', block: 'center' });
    });
  });
})();
</script>
